26  /**
27   * RuleLexerToken uses RuledBasedTextCategorizer in PREFIX_MODE for splitting
28   * the input stream into words. Then it calls a SentenceDectector to group them
29   * together in sentences. It can share categorizers and sentence detectors with
30   * other pipeline jobs when they are thread-safe, since they may load big
31   * dictionaries.
32   */
33  @Component(
34  	name = "ruled-lex-service",
35  	service = { LexService.class }
36  )
37  public class RuleBasedLexer implements LexService {
38  
39  	class RuleLexerToken extends LexService.LexerToken {
40  		private Map<Locale, ISentenceDetector> mSentDetectors = new HashMap<Locale, ISentenceDetector>();
41  		private Map<Locale, RuleBasedTextCategorizer> mTextCategorizers = new HashMap<Locale, RuleBasedTextCategorizer>();
42  		private RuleBasedTextCategorizer mGenericCategorizer;
43  		private ISentenceDetector mGenericSentDetector;

45  		public RuleLexerToken(RuleBasedLexer l) {
46  			super(l);

49  		@Override
50  		public List<Sentence> split(String input, Locale lang, List<String> parsingErrors) {
51  			if (input.length() == 0)
52  				return Collections.EMPTY_LIST;
53  
54  			RuleBasedTextCategorizer categorizer = mTextCategorizers.get(lang);
55  			ISentenceDetector splitter = mSentDetectors.get(lang);
56  
57  			// call the categorizer and the sentence detector
58  			List<CategorizedWord> words = splitIntoWords(input, lang, categorizer);
59  
60  			for (int k = 0; k < words.size(); ++k) {
61  				if (words.get(k).category == Category.UNKNOWN) {
62  					StringBuilder error = new StringBuilder(getName()
63  					        + ": the lexeme between square brackets could not be recognized: ");
64  					for (int i = Math.max(0, k - 10); i < k; ++i)
65  						error.append(words.get(i).word);
66  					error.append("[" + words.get(k).word + "]");
67  					for (int i = k + 1; i < Math.min(words.size(), k + 10); ++i)
68  						error.append(words.get(i).word);
69  					parsingErrors.add(error.toString());
70  				}
71  			}
72  
73  			if (words.size() == 0
74  			        || (words.size() == 1 && !TextCategorizer.isSpeakable(words.iterator()
75  			                .next().category)))
76  				return Collections.EMPTY_LIST;
77  
78  			List<List<CategorizedWord>> sentences = splitter.split(words);
79  
80  			// build the sentences in the expected format
81  			int currentPos = 0;
82  			List<Sentence> res = new ArrayList<Sentence>();
83  			for (List<CategorizedWord> sentence : sentences) {
84  				//discard empty sentences
85  				int emptySize = 0;
86  				for (CategorizedWord word : sentence) {
87  					if (TextCategorizer.isSpeakable(word.category)) {
88  						emptySize = -1;
89  						break;
90  					}
91  					emptySize += word.word.length();
92  				}
93  				if (emptySize != -1) {
94  					currentPos += emptySize;
95  					continue;
96  				}
97  
98  				Sentence s = new Sentence();
99  				s.boundaries = new TextBoundaries();
100 				res.add(s);
101 
102 				//NOTE: Now, the StringComposer should already trim the sentences 
103 
104 				//find the beginning of the sentence
105 				ListIterator<CategorizedWord> it = sentence.listIterator();
106 				while (it.hasNext()) {
107 					CategorizedWord word = it.next();
108 					if (word.category != Category.SPACE) {
109 						s.boundaries.left = currentPos;
110 						it.previous();
111 						break;
112 					}
113 					currentPos += word.word.length();
114 				}
115 
116 				//content
117 				s.words = new ArrayList<LexService.TextBoundaries>();
118 				while (it.hasNext()) {
119 					CategorizedWord word = it.next();
120 					if (TextCategorizer.isSpeakable(word.category)) {
121 						TextBoundaries bounds = new TextBoundaries();
122 						bounds.left = currentPos;
123 						bounds.right = bounds.left + word.word.length();
124 						s.words.add(bounds);
125 					}
126 					currentPos += word.word.length();
127 				}
128 
129 				//go backward to find the end of the sentence
130 				int end = currentPos;
131 				while (it.hasPrevious()) {
132 					CategorizedWord word = it.previous();
133 					if (word.category != Category.SPACE) {
134 						s.boundaries.right = end;
135 						break;
136 					}
137 					end -= word.word.length();
138 				}
139 			}
140 
141 			return res;
142 		}

144 		private List<CategorizedWord> splitIntoWords(String input, Locale locale,
145 		        RuleBasedTextCategorizer categorizer) {
146 			String lowerCase = input.toLowerCase(locale);
147 			LinkedList<CategorizedWord> result = new LinkedList<CategorizedWord>();
148 
149 			int shift = 0;
150 			while (shift < input.length()) {
151 				String right = input.substring(shift);
152 				String lowerCaseRight = lowerCase.substring(shift);
153 				CategorizedWord w = categorizer.categorize(right, lowerCaseRight);
154 				result.add(w);
155 				shift += w.word.length();
156 			}
157 
158 			return result;
159 		}

161 		@Override
162 		public void shareResourcesWith(LexerToken other, Locale lang) {
163 			RuleLexerToken rother = (RuleLexerToken) other;
164 
165 			if (mGenericCategorizer == null) {
166 				RuleBasedTextCategorizer genericCategorizer = rother.mGenericCategorizer;
167 				if (genericCategorizer != null && genericCategorizer.threadsafe()) {
168 					mGenericCategorizer = genericCategorizer;
169 				}
170 			}
171 
172 			if (mGenericSentDetector == null) {
173 				ISentenceDetector genericSentDetector = rother.mGenericSentDetector;
174 				if (genericSentDetector != null && genericSentDetector.threadsafe()) {
175 					mGenericSentDetector = genericSentDetector;
176 				}
177 			}
178 
179 			if (mTextCategorizers.get(lang) == null) {
180 				RuleBasedTextCategorizer categorizer = rother.mTextCategorizers.get(lang);
181 				if (categorizer != null && categorizer.threadsafe()) {
182 					mTextCategorizers.put(lang, categorizer);
183 				}
184 			}
185 
186 			if (mSentDetectors.get(lang) == null) {
187 				ISentenceDetector sentDetector = rother.mSentDetectors.get(lang);
188 				if (sentDetector != null && sentDetector.threadsafe()) {
189 					mSentDetectors.put(lang, sentDetector);
190 				}
191 			}
192 		}

194 		@Override
195 		public void addLang(Locale lang) throws LexerInitException {
196 			if (mGenericSentDetector == null) {
197 				mGenericSentDetector = new EuroSentenceDetector();
198 			}
199 
200 			if (mSentDetectors.get(lang) == null) {
201 				mSentDetectors.put(lang, mGenericSentDetector);
202 			}
203 
204 			try {
205 				if (mGenericCategorizer == null) {
206 					mGenericCategorizer = new RuledMultilangCategorizer();
207 					mGenericCategorizer.init(MatchMode.PREFIX_MATCH);
208 					mGenericCategorizer.compile();
209 				}
210 
211 				if (mTextCategorizers.get(lang) == null) {
212 					String iso639_2lang = lang.getISO3Language();
213 					RuleBasedTextCategorizer rtc = mGenericCategorizer;
214 					if ("fre".equals(iso639_2lang) || "fra".equals(iso639_2lang)
215 					        || "frm".equals(iso639_2lang) || "fro".equals(iso639_2lang)) {
216 						rtc = new RuledFrenchCategorizer();
217 						rtc.init(MatchMode.PREFIX_MATCH);
218 						rtc.compile();
219 					}
220 
221 					mTextCategorizers.put(lang, rtc);
222 				}
223 
224 			} catch (IOException e) {
225 				throw new LexerInitException(e.getCause());
226 			}
227 		}

231 	@Override
232 	public int getLexQuality(Locale lang) {
233 		String language = lang.getLanguage();
234 
235 		if (language.equals(new Locale("fr").getLanguage()))
236 			return 3 * LexService.MinSpecializedLexQuality;
237 
238 		if (language.equals(new Locale("en").getLanguage()))
239 			return 3 * LexService.MinSpecializedLexQuality;
240 
241 		if (okLanguages.contains(language)) {
242 			return 2 * LexService.MinSpecializedLexQuality;
243 		}
244 
245 		return 0;
246 	}

1   package org.daisy.pipeline.nlp.lexing.ruled.impl;
2   
3   import java.util.LinkedList;
4   import java.util.List;
5   import java.util.regex.Pattern;
6   
7   import org.daisy.pipeline.nlp.TextCategorizer.CategorizedWord;
8   import org.daisy.pipeline.nlp.TextCategorizer.Category;
9   
10  /**
11   * Dummy sentence detector based on the detection of combination of periods,
12   * spaces and capitalized words. The class is thread-safe.
13   */
14  public class BasicSentenceDetector implements ISentenceDetector {
15  	private Pattern mStrongDelimiters;
16  	private Pattern mWeakDelimiters;
17  
18  	public BasicSentenceDetector() {
19  		mWeakDelimiters = Pattern.compile("[.:]+");
20  		mStrongDelimiters = Pattern.compile("[.:?!]*[?!…]+[.:?!]*");
21  	}
22  
23  	@Override
24  	public List<List<CategorizedWord>> split(List<CategorizedWord> CategorizedWords) {
25  
26  		List<List<CategorizedWord>> result = new LinkedList<List<CategorizedWord>>();
27  		List<CategorizedWord> currentSentence = new LinkedList<CategorizedWord>();
28  
29  		int delimiter = 0; // 2: sure, 1: possible, 0: no
30  		for (CategorizedWord w : CategorizedWords) {
31  			if (currentSentence.size() > 0
32  			        && w.category != Category.PUNCTUATION
33  			        && w.category != Category.SPACE
34  			        && (delimiter == 2 || (delimiter == 1 && w.category == Category.COMMON && Character
35  			                .isUpperCase(w.word.charAt(0))))) {
36  				result.add(currentSentence);
37  				currentSentence = new LinkedList<CategorizedWord>();
38  			}
39  
40  			currentSentence.add(w);
41  
42  			if (mStrongDelimiters.matcher(w.word).matches()) {
43  				delimiter = 2;
44  			} else if (mWeakDelimiters.matcher(w.word).matches()) {
45  				delimiter = 1;
46  			} else if (w.category != Category.SPACE)
47  				delimiter = 0;
48  		}
49  
50  		if (currentSentence.size() > 0)
51  			result.add(currentSentence);
52  
53  		return result;
54  	}
55  
56  	@Override
57  	public boolean threadsafe() {
58  		return true;
59  	}
60  }

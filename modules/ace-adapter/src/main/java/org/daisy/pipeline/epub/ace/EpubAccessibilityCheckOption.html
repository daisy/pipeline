1   package org.daisy.pipeline.epub.ace;
2   
3   import java.util.ListResourceBundle;
4   import java.util.NoSuchElementException;
5   
6   import org.osgi.framework.Version;
7   
8   public class EpubAccessibilityCheckOption extends ListResourceBundle {
9   
10  	private static final String latestAceVersion = "1.4.6";
11  
12  	private final Object[][] resources;
13  
14  	public EpubAccessibilityCheckOption() {
15  		super();
16  		String name = "Enable accessibility check";
17  		String desc = "Check the compliance to the EPUB accessibility specification using the [DAISY Ace](https://daisy.github.io/ace) tool.";
18  		try {
19  			Ace ace = AceFinder.get();
20  			try {
21  				Version version = Version.parseVersion(ace.getVersion());
22  				desc += String.format("\n\nAce version %s was found on your system.", version);
23  				if (version.compareTo(Version.parseVersion(latestAceVersion)) < 0)
24  					desc += String.format(" A newer version is available. Check [how to update Ace]("
25  					                      + "https://daisy.github.io/ace/getting-started/installation/#install-or-update-ace).");
26  			} catch (IllegalArgumentException e) {
27  			}
28  		} catch (NoSuchElementException e) {
29  			desc += "\n\nThis option can currently not be used because Ace was not found on your system.";
30  			desc += " Check [how to install Ace](https://daisy.github.io/ace/getting-started/installation/) on your system.";
31  		}
32  		desc += "\n\nThis option is only available for zipped EPUBs.";
33  		resources = new Object[][]{
34  			{ "epub3-validator.accessibility-check.name", name },
35  			{ "epub3-validator.accessibility-check.desc", desc }
36  		};
37  	}
38  
39  	@Override
40  	protected Object[][] getContents() {
41  		return resources;
42  	}
43  }
